# Пример карты со своими текстурами у кубов.
#
# Формат строки куба:  x y z sx sy sz [tex=<файл>] [tile=<метры>] [repeat|stretch]
# Директивы (действуют на все кубы ниже):
#   texture <файл>   — текстура по умолчанию
#   tile <метры>     — размер одной копии текстуры, 0 — как в движке (2 м)
#   repeat           — рисунок продолжается по поверхности (по умолчанию)
#   stretch          — одна копия растянута на каждую грань целиком
#
# Пути относительны файлу карты, затем текущего каталога; кавычки — для
# пробелов в имени файла. Куб без своей текстуры рисуется текстурой
# по умолчанию, а если её в карте нет — текстурой, переданной при запуске.
#
# Запуск: game3d texture.raw examples/map_with_textures.txt

# --- Общая земля: камень, одна копия 4 x 4 метра ---------------------------
texture stone.raw
tile 4

-16 0 -16 32 1 32

# --- Дом: стены из кирпича, пол и потолок из досок -------------------------
texture plank.raw
tile 2

-6 1 -6 12 1 12      # пол  (текстура по умолчанию — доски)
-6 6 -6 12 1 12      # потолок

texture brick.raw
tile 1

-6 2 -6 12 4 0.5     # задняя стена
-6 2  5.5 12 4 0.5   # передняя стена
-6 2 -6 0.5 4 12     # левая стена
 5.5 2 -6 0.5 4 12   # правая стена

# --- Простенок: та же текстура, но уложена крупнее --------------------------
# tile= перебивает директиву только для этого куба: одна копия 2 x 2 метра
-6 2 -0.5 2 4 1 tex=brick.raw tile=2

# --- Отдельные кубы со своей текстурой -------------------------------------
texture metal.raw
tile 1

20 1 0 1 2 1 tex=metal.raw          # столб (текстура = директива выше)
24 1 0 2 2 2 tex="metal rough.raw"  # имя файла с пробелом — в кавычках

# --- Табличка: рисунок виден целиком, а не «плиткой» -----------------------
# stretch растягивает одну копию на всю грань; годится и для вывесок,
# и для экранов, и для картин на стене.
26.5 2.5 0.5 0.2 1 1 tex=sign.raw stretch

# --- Мост: перила досками, настил камнем -----------------------------------
texture plank.raw
tile 1

30 2 -8 6 0.3 0.3
30 2  7.7 6 0.3 0.3
texture stone.raw
tile 3
30 1.7 -8 6 0.3 16

# --- Файла текстуры нет ----------------------------------------------------
# Куб нарисуется текущей текстурой по умолчанию (здесь stone.raw), в консоль
# попадёт предупреждение с номером строки, карта загрузится дальше.
40 1 0 2 2 2 tex=missing_texture.raw
